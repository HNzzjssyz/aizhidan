五、fetchCozeToken() — 获取令牌

async function fetchCozeToken() {
  try {
    var response = await fetch('/system/bot/token.xt');
    if (!response.ok) throw new Error('令牌接口返回错误: ' + response.status);
    var data = await response.json();
    if (data.ok !== 1 || !data.accessToken) throw new Error(data.msg || '获取令牌失败');
    return data.accessToken;
  } catch (e) {
    console.error('fetchCozeToken error:', e);
    throw e;
  }
}
📘 接口说明
项目值
请求地址/system/bot/token.xt
请求方法GET
请求参数无
成功响应{"ok":1,"accessToken":"xxx"}
失败响应{"ok":0,"msg":"错误信息"}
成功判断data.ok === 1 && data.accessToken

六、initCozeSDK() — 初始化 SDK

async function initCozeSDK() {
  var btn = document.getElementById('connectCozeBtn');
  btn.disabled = true;
  btn.innerHTML = '<i class="ri-loader-4-line" style="font-size:16px;animation:spin 1s linear infinite;"></i><span>获取授权...</span>';
  document.getElementById('apiStatus').textContent = '正在获取授权...';

  try {
    cozeAccessToken = await fetchCozeToken();

    if (typeof CozeWebSDK === 'undefined') {
      throw new Error('Coze SDK 未加载，请刷新页面');
    }

    if (cozeClient && typeof cozeClient.destroy === 'function') {
      cozeClient.destroy();
    }

    cozeClient = new CozeWebSDK.WebChatClient({
      config: {
        bot_id: '7638933573109792777',
      },
      auth: {
        type: 'token',
        token: cozeAccessToken,
        onRefreshToken: async function () {
          try {
            cozeAccessToken = await fetchCozeToken();
            return cozeAccessToken;
          } catch (e) {
            console.error('onRefreshToken error:', e);
            return cozeAccessToken;
          }
        }
      },
      ui: {
        base: { lang: 'zh-CN', zIndex: 50 },
        asstBtn: { isNeed: true },
        chatBot: {
          title: '智能制单助手',
          uploadable: true,
          width: 460,
        },
        header: { isShow: true, isNeedClose: false }
      }
    });

    setTimeout(function() {
      if (cozeClient && typeof cozeClient.showChatBot === 'function') {
        cozeClient.showChatBot();
      }
    }, 500);

    document.getElementById('apiStatus').textContent = '已连接 ✅';
    document.getElementById('apiStatus').style.color = '#16a34a';
    btn.disabled = false;
    btn.innerHTML = '<i class="ri-check-line" style="font-size:14px;"></i><span>已连接</span>';
    btn.style.background = '#16a34a';
    showToast('智能体已启动');

  } catch (e) {
    console.error('Coze SDK 初始化失败:', e);
    btn.disabled = false;
    btn.innerHTML = '<i class="ri-play-line" style="font-size:16px;"></i><span>启动智能体</span>';
    btn.style.background = '';
    document.getElementById('apiStatus').textContent = e.message;
    document.getElementById('apiStatus').style.color = '#e5484d';
    showToast('启动失败: ' + e.message);
  }
}

6.1 初始化流程

  1. 禁用按钮，显示加载动画
  2. 调用 fetchCozeToken() 获取 access_token
  3. 检查 CozeWebSDK 全局对象是否存在
  4. 若已有客户端实例，先调用 destroy() 销毁
  5. 创建 CozeWebSDK.WebChatClient 实例
  6. 500ms 后调用 showChatBot() 弹出聊天窗口
  7. 更新按钮和状态为"已连接"

6.4 Token 自动续期

SDK 通过 onRefreshToken 回调实现 token 自动续期：

  • 当 access_token 过期时，SDK 自动调用 onRefreshToken
  • 回调内重新调用 fetchCozeToken() 获取新 token
  • 成功时返回新 token，失败时返回旧 token（降级保活）

七、isInCozeSDKArea() — SDK 区域检测

function isInCozeSDKArea(el) {
  while (el) {
    if (el.className && typeof el.className === 'string') {
      if (el.className.indexOf('coze-chat-sdk') !== -1 || el.className.indexOf('coz-') !== -1) {
        return true;
      }
    }
    el = el.parentElement;
  }
  return false;
}

通过 DOM 类名向上遍历，判断目标元素是否在 Coze SDK 聊天窗口内。匹配规则：

  • 类名包含 coze-chat-sdk
  • 类名以 coz- 为前缀

八、injectImageToCozeSDK() — 图片粘贴注入

function injectImageToCozeSDK(e) {
  try {
    var items = e.clipboardData && e.clipboardData.items;
    if (!items) return;

    var hasImage = false;
    for (var i = 0; i < items.length; i++) {
      if (items[i].type.indexOf('image') !== -1) {
        hasImage = true;
        break;
      }
    }
    if (!hasImage) return;

    if (!isInCozeSDKArea(e.target)) return;

    var fileInput = document.querySelector('input.coze-chat-sdk-semi-upload-hidden-input');
    if (!fileInput) return;

    e.preventDefault();
    e.stopPropagation();

    var files = [];
    for (var i = 0; i < items.length; i++) {
      if (items[i].type.indexOf('image') !== -1) {
        var file = items[i].getAsFile();
        if (file) files.push(file);
      }
    }
    if (files.length === 0) return;

    var dataTransfer = new DataTransfer();
    files.forEach(function(f) { dataTransfer.items.add(f); });
    fileInput.files = dataTransfer.files;
    fileInput.dispatchEvent(new Event('change', { bubbles: true }));
  } catch (e) {
    console.warn('injectImageToCozeSDK error:', e);
  }
}

document.addEventListener('paste', injectImageToCozeSDK, true);

8.1 执行流程

  1. 监听全局 paste 事件（捕获阶段）
  2. 检查剪贴板是否包含图片类型数据
  3. 检查粘贴目标是否在 Coze SDK 区域内（isInCozeSDKArea）
  4. 查找 SDK 内部的隐藏文件输入框 input.coze-chat-sdk-semi-upload-hidden-input
  5. 拦截默认粘贴行为（preventDefault + stopPropagation）
  6. 通过 DataTransfer API 构造文件列表
  7. 将文件列表赋值给隐藏输入框，触发 change 事件完成上传

8.2 作用域限制

📘 仅在 SDK 区域内生效

通过 isInCozeSDKArea(e.target) 判断，只有当粘贴操作发生在 Coze 聊天窗口内时才拦截。页面其他区域（如订单 JSON 粘贴区）的粘贴行为不受影响。
